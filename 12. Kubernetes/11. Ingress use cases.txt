Use cases:
    
    1. Multiple paths:
        - We can host multiple application on same path with different host.
        - We can transfer request to separate service based on path

        ex. google.com/maps     google.com/search

        apiVersion: networking.k8s.io/v1
        kind: Ingress
        metadata:
            name: app-ingress
        spec:
            rules:
            - host: app.local
                http:
                paths:
                - path: /
                    pathType: Prefix
                    backend:
                    service:
                        name: frontend-service
                        port:
                        number: 80

                - path: /api
                    pathType: Prefix
                    backend:
                    service:
                        name: backend-service
                        port:
                        number: 80

                - path: /admin
                    pathType: Prefix
                    backend:
                    service:
                        name: admin-service
                        port:
                        number: 80

    2. Sub domains:
    
        apiVersion: networking.k8s.io/v1
        kind: Ingress
        metadata:
            name: multi-subdomain-ingress
            namespace: default
        spec:
            rules:
            - host: api.app.local
                http:
                paths:
                - path: /
                    pathType: Prefix
                    backend:
                    service:
                        name: backend-service
                        port:
                        number: 80
            - host: admin.app.local
                http:
                paths:
                - path: /
                    pathType: Prefix
                    backend:
                    service:
                        name: admin-service
                        port:
                        number: 80
