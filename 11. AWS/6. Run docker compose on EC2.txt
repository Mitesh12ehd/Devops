Using docker compose for deployment:
    
    1. Install docker compose on EC2:
        - we need to install docker compose apart from docker
        - docker compose not available in yum

        1. download executable of docker compose
            sudo curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose

        2. Add execute permission 
            sudo chmod +x /usr/local/bin/docker-compose

    2. Available docker compose file on EC2:
        - we need to copy docker compose file on EC2

    3. Modify Jenkinsfile

        docker-compose.yaml
            version: "3.8"

            services:
            java-maven-app:
                image: miteshch/demo-app:java-maven-1.0
                ports:
                - 8080:8080
            postgres:
                image: postgres:15
                ports:
                - 5432:5432
                environment:
                - POSTGRES_PASSWORD=my-pwd

        Jenkinsfile
            environment{
                IMAGE_NAME = "miteshch/demo-app:java-maven-1.0"
            }
            stage("deploy"){
                steps{
                    script{ 
                        echo "Deploying docker image to EC2..."
                        def dockerCommand = "docker-compose -f docker-compose.yaml up --detach"
                        sshagent(['ec2-server-key']) {
                            // Copy docker compose file into working directory of EC2
                            sh "scp docker-compose.yaml ec2-user@13.201.190.56:/home/ec2-user"
                            // -o flag to avoid popup that ask for yes when we connect using ssh
                            sh "ssh -o StrictHostKeyChecking=no ec2-user@13.201.190.56 ${dockerCommand}"
                        }
                    }
                }
            }

    4. Improve code by using shell script to write command:
        - Create shell file and copy into EC2
        - execute shell file
        - Pass image name in docker compose to make dynamic image name
        
        server-cmds.sh
            #!/user/bin/env bash
            export IMAGE_NAME=$1
            docker-compose -f docker-compose.yaml up --detach 

        Jenkinsfile
            environment{
                IMAGE_NAME = "miteshch/demo-app:java-maven-1.0"
            }
            stage("deploy"){
                steps{
                    script{ 
                        echo "Deploying docker image to EC2..."
                        def shellCmd = "bash ./server-cmds.sh ${IMAGE_NAME}"
                        sshagent(['ec2-server-key']) {
                            // Copy docker compose and shell file on EC2
                            sh "scp docker-compose.yaml ec2-user@13.201.190.56:/home/ec2-user"
                            sh "scp server-cmds.sh ec2-user@13.201.190.56:/home/ec2-user"

                            // -o flag to avoid popup that ask for yes when we connect using ssh
                            sh "ssh -o StrictHostKeyChecking=no ec2-user@13.201.190.56 ${shellCmd} "
                        }
                    }
                }
            }

        docker-compose.yaml
            version: "3.8"

            services:
            java-maven-app:
                image: {IMAGE_NAME}
                ports:
                - 8080:8080
            postgres:
                image: postgres:15
                ports:
                - 5432:5432
                environment:
                - POSTGRES_PASSWORD=my-pwd
